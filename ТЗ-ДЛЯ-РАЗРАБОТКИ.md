## БАЗА ДАННЫХ (Prisma)

```prisma
// prisma/schema.prisma

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

// ========== ПОЛЬЗОВАТЕЛИ ==========
model User {
  id            String    @id @default(cuid())
  email         String    @unique
  passwordHash  String
  role          String    @default("user") // user | teacher | admin
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  // Отношения
  subscriptions Subscription[]
  payments      Payment[]
  generatorUses GeneratorUse[]
  
  @@map("users")
}

// ========== ПОДПИСКА ==========
model Subscription {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  
  status        String    @default("active") // active | cancelled | expired
  startDate     DateTime  @default(now())
  endDate       DateTime
  autoRenew     Boolean   @default(true)
  paymentId     String?   // ID платежа в ЮKassa
  
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  @@unique([userId])
  @@map("subscriptions")
}

// ========== ПЛАТЕЖИ ==========
model Payment {
  id            String    @id @default(cuid())
  userId        String
  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  
  kassaId       String    @unique // ID платежа в ЮKassa
  amount        Float
  currency      String    @default("RUB")
  status        String    // pending | succeeded | failed
  
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  @@map("payments")
}

// ========== КОНТЕНТ ==========
model Segment {
  id        String    @id @default(cuid())
  slug      String    @unique // 4-5-let, 6-7-let, 1-klass
  name      String    // "Дошкольники 4–5 лет"
  
  subjects  Subject[]
  topics    Topic[]
  
  @@map("segments")
}

model Subject {
  id        String    @id @default(cuid())
  slug      String    // matematika, russkiy, angliyskiy
  name      String
  icon      String?   // эмодзи или иконка
  
  segmentId String
  segment   Segment   @relation(fields: [segmentId], references: [id], onDelete: Cascade)
  
  topics    Topic[]
  
  @@unique([slug, segmentId])
  @@map("subjects")
}

model Topic {
  id          String    @id @default(cuid())
  slug        String
  title       String
  description String?
  
  segmentId   String
  segment     Segment   @relation(fields: [segmentId], references: [id], onDelete: Cascade)
  
  subjectId   String
  subject     Subject   @relation(fields: [subjectId], references: [id], onDelete: Cascade)
  
  formats     Format[]
  
  @@unique([slug, subjectId, segmentId])
  @@map("topics")
}

model Format {
  id        String    @id @default(cuid())
  type      String    // teoria | shpargalka | test | trenazher | igra | kartochki | generator | plakat | vpr | material_teacher
  content   String?   // HTML или markdown
  
  topicId   String
  topic     Topic     @relation(fields: [topicId], references: [id], onDelete: Cascade)
  
  isPublic  Boolean   @default(true) // для теории и шпаргалок
  isPaid    Boolean   @default(false) // платный контент
  
  createdAt DateTime  @default(now())
  
  @@unique([topicId, type])
  @@map("formats")
}

// ========== ГЕНЕРАТОР ==========
model GeneratorUse {
  id        String    @id @default(cuid())
  userId    String?   // null для гостей (используем sessionId вместо)
  user      User?     @relation(fields: [userId], references: [id], onDelete: SetNull)
  
  sessionId String?   // для гостей
  type      String    // primery | propisi | krosswordy | diktanty | izlozheniya
  
  createdAt DateTime  @default(now())
  
  @@map("generator_uses")
}

// ========== АНАЛИТИКА ==========
model PageView {
  id        String    @id @default(cuid())
  url       String
  userId    String?
  
  createdAt DateTime  @default(now())
  
  @@map("page_views")
}

model TopicAccess {
  id        String    @id @default(cuid())
  topicId   String
  userId    String?
  
  createdAt DateTime  @default(now())
  
  @@map("topic_accesses")
}
```

```
POST /api/auth/register
Body: { email, password, agreeToTerms }
Response: { id, email, token }

POST /api/auth/login
Body: { email, password }
Response: { id, email, token, subscription: { status, endDate } }

POST /api/auth/logout
Response: { success: true }
```

```
GET /api/content/topics?segment=4-5-let&subject=matematika
Response: [{ id, slug, title, formats: [{ type, isPaid, isPublic }] }]

GET /api/content/[id]
Response: { id, slug, title, description, formats: { teoria, shpargalka, ... } }

GET /api/content/search?q=счёт
Response: [{ id, title, segment, subject }]
```

```
POST /api/payment/create
Body: { userId, amount: 29900 }
Response: { kassaId, redirectUrl, expiresAt }

POST /api/payment/webhook (от ЮKassa)
Body: { object: { id, status, ... } }
Response: { success: true }
```

### ГЕНЕРАТОР

```
POST /api/generator/create
Body: { type: "primery", params: { min: 0, max: 10, count: 20 } }

```

import jwt from 'jsonwebtoken';
import bcrypt from 'bcryptjs';

export function generateToken(userId: string) {
  return jwt.sign({ userId }, process.env.JWT_SECRET, { expiresIn: '30d' });
}

export function verifyToken(token: string) {
  try {
    return jwt.verify(token, process.env.JWT_SECRET);
  } catch {
    return null;
  }
}

export async function hashPassword(password: string) {
  return bcrypt.hash(password, 10);
}

export async function comparePassword(password: string, hash: string) {
  return bcrypt.compare(password, hash);
}

## ИНТЕГРАЦИЯ ЮKASSA

```typescript
// lib/kassa.ts

import YooKassa from 'yookassa';

const kassa = new YooKassa({
  shopId: process.env.NEXT_PUBLIC_KASSA_SHOP_ID,
  secretKey: process.env.KASSA_SECRET_KEY,
});

export async function createPayment(userId: string, amount: number) {
  const payment = await kassa.createPayment({
    amount: { value: (amount / 100).toFixed(2), currency: 'RUB' },
    confirmation: {
      type: 'redirect',
      return_url: `${process.env.NEXT_PUBLIC_SITE_URL}/payment-success`,
    },
    metadata: { userId },
  });

  return {
    kassaId: payment.id,
    redirectUrl: payment.confirmation.confirmation_url,
  };
}

export async function confirmPayment(kassaId: string) {
  const payment = await kassa.getPayment(kassaId);
  return payment.status === 'succeeded';
}
```
